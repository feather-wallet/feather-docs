The inner is tagged so the page template can distinguish it from inline . */ -}} {{- $lang := cond (eq .Type "") "plaintext" .Type -}} {{- $code := .Inner | strings.TrimRight "\n" -}} {{- $code = replace $code "&" "&" -}} {{- $code = replace $code "<" "<" -}} {{- $code = replace $code ">" ">" -}} {{- printf `
%s` $lang $code | safeHTML }}
